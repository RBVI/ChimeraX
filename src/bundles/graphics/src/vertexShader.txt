#if defined(USE_IMPOSTOR_SPHERE) || defined(USE_IMPOSTOR_CYLINDER)
// Each instance is drawn as two triangles per quad, without vertex buffers.
// Quad corners come from the vertex index, and for dashed cylinders the
// quad index selects the dash.
const vec2 impostor_corners[6] = vec2[6](vec2(-1,-1), vec2(1,-1), vec2(1,1),
                                         vec2(-1,-1), vec2(1,1), vec2(-1,1));
vec2 impostor_corner()
{
  return impostor_corners[gl_VertexID % 6];
}

// Perspective slope bounds for a sphere wholly in front of the eye.  These
// also bound a cylinder end disk, without magnifying the entire bond by the
// depth of its nearest endpoint.  xy can be expressed in any screen basis.
void sphere_slope_bounds(vec3 center, float radius,
                         out vec2 lower, out vec2 upper)
{
  float depth = -center.z;
  float denom = (depth - radius) * (depth + radius);
  vec2 root = sqrt(max(vec2(denom) + center.xy * center.xy, vec2(0.0)));
  lower = (center.xy * depth - radius * root) / denom;
  upper = (center.xy * depth + radius * root) / denom;
}

vec4 near_clipped_box_proxy(vec3 lower, vec3 upper)
{
  // Proxy for geometry within an eye-space box crossing the eye plane.  Its
  // slope bounds are unbounded, but only the part beyond the near plane is
  // drawn (as with triangles), which projects to a finite rectangle.  Covering
  // the viewport instead made every atom within a radius of the (infinite) eye
  // plane shade the full screen up close.  Slope xy/-z is monotonic in x, y
  // and z over the box, so its extremes are at the box corners.
  mat4 p = projection_matrix;
  float near_z = (p[3][2] + p[3][3]) / (-p[2][3] - p[2][2]);
  if (lower.z > near_z)
    return vec4(0.0, 0.0, 1.0, 1.0); // Entirely in front of the near plane: clipped.
  float front_z = min(upper.z, near_z);
  vec2 slope_lower = min(lower.xy / -lower.z, lower.xy / -front_z);
  slope_lower = min(slope_lower, min(upper.xy / -lower.z, upper.xy / -front_z));
  vec2 slope_upper = max(lower.xy / -lower.z, lower.xy / -front_z);
  slope_upper = max(slope_upper, max(upper.xy / -lower.z, upper.xy / -front_z));
  vec2 slope = 0.5 * (slope_lower + slope_upper) +
               impostor_corner() * 0.5025 * (slope_upper - slope_lower);
  return vec4(slope * -front_z, front_z, 1.0);
}
#endif

void main(void)
{
#if defined(USE_IMPOSTOR_SPHERE) || defined(USE_IMPOSTOR_CYLINDER)
  // The same affine instance placements used by triangle geometry.
  #ifdef USE_SHIFT_AND_SCALE
    float scale = instance_shift_and_scale.w;
    mat4 primitive_placement = mat4(vec4(scale, 0, 0, 0), vec4(0, scale, 0, 0),
      vec4(0, 0, scale, 0), vec4(instance_shift_and_scale.xyz, 1));
  #elif defined(USE_INSTANCING)
    mat4 primitive_placement = instance_placement;
  #else
    mat4 primitive_placement = mat4(1);
  #endif
#endif
#ifdef USE_IMPOSTOR_SPHERE
  // Impostor sphere: expand unit quad into camera-facing billboard
  vec3 center_eye = (model_view_matrix * primitive_placement[3]).xyz;
  float radius = length((model_view_matrix * primitive_placement[0]).xyz);
  if (radius <= 0.0) {
    gl_Position = vec4(2, 2, 2, 1);
    return;
  }
  bool sphere_ortho = (projection_matrix[3][3] != 0.0);
  vec4 veye;
  if (sphere_ortho) {
    veye = vec4(center_eye + vec3(impostor_corner() * radius * 1.005, radius), 1.0);
  } else {
    // Smallest axis-aligned rectangle enclosing the perspective projection of
    // the sphere.  Unlike a symmetric billboard, its bounds shift off-axis.
    float depth = -center_eye.z;
    if (depth > radius * 1.000001) {
      vec2 slope_min, slope_max;
      sphere_slope_bounds(center_eye, radius, slope_min, slope_max);
      vec2 slope_mid = 0.5 * (slope_min + slope_max);
      vec2 slope_half = 0.5025 * (slope_max - slope_min);
      vec2 slope = slope_mid + impostor_corner() * slope_half;
      // Place the quad at the front of the sphere so the far clip plane
      // does not remove it while the front of the sphere is in view.
      float front_z = center_eye.z + radius;
      veye = vec4(slope * -front_z, front_z, 1.0);
    } else if (depth < -radius) {
      veye = vec4(0.0, 0.0, 1.0, 1.0); // Entirely behind the eye: clipped.
    } else
      veye = near_clipped_box_proxy(center_eye - radius, center_eye + radius);
  }
  impostor_center = center_eye;
  impostor_radius = radius;

  #ifdef USE_CLIP_PLANES
    for (int i = 0 ; i < num_clip_planes ; ++i)
      gl_ClipDistance[i] = 1.0; // Analytic surface clipping is done per fragment.
  #endif

#elif defined(USE_IMPOSTOR_CYLINDER)
  // Unit cylinder endpoints are transformed by ordinary affine placements.
  // The quad index selects a dash, matching the existing
  // dashed_cylinder_geometry generator exactly.
  float unit_height = impostor_cylinder.x;
  float segments = impostor_cylinder.z;
  float dash_center = segments > 0.0 ?
    (float(gl_VertexID / 6) - 0.5 * (segments - 1.0)) * unit_height / segments : 0.0;
  float dash_half_height = segments > 0.0 ? unit_height / (4.0 * segments) : unit_height / 2.0;
  vec4 cylinder_start = primitive_placement * vec4(0, 0, dash_center - dash_half_height, 1);
  vec4 cylinder_end = primitive_placement * vec4(0, 0, dash_center + dash_half_height, 1);
  vec3 p0_eye = (model_view_matrix * cylinder_start).xyz;
  vec3 p1_eye = (model_view_matrix * cylinder_end).xyz;
  vec3 cyl_axis_eye = p1_eye - p0_eye;
  float cyl_len = length(cyl_axis_eye);
  vec3 cyl_dir = cyl_axis_eye / max(cyl_len, 1e-7);
  float cyl_radius = length((model_view_matrix * primitive_placement[0]).xyz);
  if (cyl_radius <= 0.0 || cyl_len <= 1e-7) {
    gl_Position = vec4(2, 2, 2, 1);
    return;
  }
  vec3 cyl_center = (p0_eye + p1_eye) * 0.5;
  float cyl_half_len = cyl_len * 0.5;

  // Use a screen-aligned plane and orient its rectangle along the projected
  // bond.  The projection of the capsule is the convex hull of its endpoint
  // spheres, so their bounds enclose the finite cylinder (including caps).
  // A single bounding-sphere magnification made close bonds cover huge areas.
  bool cyl_ortho = (projection_matrix[3][3] != 0.0);
  vec4 veye;
  if (cyl_ortho || max(p0_eye.z, p1_eye.z) < -cyl_radius * 1.000001) {
    vec2 along = cyl_ortho ? p1_eye.xy - p0_eye.xy :
      p1_eye.xy / -p1_eye.z - p0_eye.xy / -p0_eye.z;
    float along_length = length(along);
    along = along_length > 1e-7 ? along / along_length : vec2(0.0, 1.0);
    vec2 across = vec2(along.y, -along.x);
    vec2 lower, upper;
    if (cyl_ortho) {
      vec2 center = vec2(dot(cyl_center.xy, across), dot(cyl_center.xy, along));
      vec2 axis = vec2(dot(cyl_dir.xy, across), dot(cyl_dir.xy, along));
      vec2 extent = cyl_half_len * abs(axis) +
        cyl_radius * sqrt(max(vec2(1.0) - axis * axis, vec2(0.0)));
      lower = center - extent;
      upper = center + extent;
    } else {
      vec2 lower0, upper0, lower1, upper1;
      sphere_slope_bounds(vec3(dot(p0_eye.xy, across), dot(p0_eye.xy, along), p0_eye.z),
                          cyl_radius, lower0, upper0);
      sphere_slope_bounds(vec3(dot(p1_eye.xy, across), dot(p1_eye.xy, along), p1_eye.z),
                          cyl_radius, lower1, upper1);
      lower = min(lower0, lower1);
      upper = max(upper0, upper1);
    }
    vec2 xy = 0.5 * (lower + upper) + impostor_corner() * 0.5025 * (upper - lower);
    // Place the quad at the front of the cylinder so the far clip plane
    // does not remove it while the front of the cylinder is in view.
    float front_z = max(p0_eye.z, p1_eye.z) + cyl_radius;
    vec2 eye_xy = (across * xy.x + along * xy.y) * (cyl_ortho ? 1.0 : -front_z);
    veye = vec4(eye_xy, front_z, 1.0);
  } else if (min(p0_eye.z, p1_eye.z) > cyl_radius) {
    veye = vec4(0.0, 0.0, 1.0, 1.0); // Entire capsule is behind the eye.
  } else
    veye = near_clipped_box_proxy(min(p0_eye, p1_eye) - cyl_radius,
                                  max(p0_eye, p1_eye) + cyl_radius);

  impostor_cyl_center = cyl_center;
  impostor_cyl_axis = cyl_dir;
  impostor_cyl_radius = cyl_radius;
  impostor_cyl_half_length = cyl_half_len;
  impostor_cyl_caps = int(impostor_cylinder.y + 0.5);

  #ifdef USE_CLIP_PLANES
    for (int i = 0 ; i < num_clip_planes ; ++i)
      gl_ClipDistance[i] = 1.0; // Analytic surface clipping is done per fragment.
  #endif

#else // not USE_IMPOSTOR_SPHERE or USE_IMPOSTOR_CYLINDER

#ifdef USE_SHIFT_AND_SCALE
  vec4 vi = vec4(instance_shift_and_scale.w * position + instance_shift_and_scale.xyz, 1);
#else
  #ifdef USE_INSTANCING
    vec4 vi = instance_placement * vec4(position, 1);
  #else
    vec4 vi = vec4(position,1);
  #endif
#endif

#ifdef USE_CLIP_PLANES
  vec4 vscene_c = model_matrix * vi;
  for (int i = 0 ; i < num_clip_planes ; ++i)
    gl_ClipDistance[i] = dot(vscene_c, clip_planes[i]);
#endif

#ifdef USE_STEREO_360
  vec4 vscene_s = model_matrix * vi;
  vec3 eye_axis = normalize(cross((vscene_s.xyz - camera_origin_and_shift.xyz),
                                  camera_vertical.xyz));
  vec4 vsoffset = vec4(vscene_s.xyz - camera_origin_and_shift.w * eye_axis, 1);
  vec4 veye = view_matrix * vsoffset;
  #ifdef USE_LIGHTING
    vshadow = (model_view_matrix * vi).xyz;
    // vshadow = (model_matrix * vi).xyz;
  #endif
#else
  vec4 veye = model_view_matrix * vi;
#endif

#endif // end USE_IMPOSTOR_SPHERE / USE_IMPOSTOR_CYLINDER

#if defined(USE_LIGHTING) || defined(USE_DEPTH_CUE) || defined(USE_IMPOSTOR_SPHERE) || defined(USE_IMPOSTOR_CYLINDER)
   v = veye.xyz;
#endif

#ifdef USE_LIGHTING
  #if !defined(USE_IMPOSTOR_SPHERE) && !defined(USE_IMPOSTOR_CYLINDER)
  #ifdef USE_LIGHTING_NORMALS
    #ifdef USE_INSTANCING
      vec3 ni = vec3(instance_placement * vec4(normal,0));
    #else
      vec3 ni = normal;
    #endif
    N = normalize((model_view_matrix * vec4(ni,0)).xyz);
  #endif
  #endif // not USE_IMPOSTOR_SPHERE/CYLINDER
  #ifdef USE_SHADOW
    #ifdef USE_STEREO_360
      shadow_tex_coord = (shadow_transform * (view_matrix * vec4(vshadow,1))).stp;
    #else
      shadow_tex_coord = (shadow_transform * veye).stp;
    #endif
  #endif
  #ifdef USE_MULTISHADOW
    //    vscene = (model_matrix * vi).xyz;
    //    nscene = (model_matrix * vec4(ni,0)).xyz;
  #endif
  #ifdef USE_TEXTURE_3D_AMBIENT
  // TODO: This should use vertex scene coords, not vertex model coordinates
    ambient_tex_coord = (ambient_tex3d_transform * vi).xyz;
  #endif
#endif // end USE_LIGHTING

#ifdef USE_VERTEX_COLORS
  color = vcolor;
#endif

#ifdef USE_TEXTURE_2D
  tex_coord_2d = tex_coord;
#endif

#ifdef USE_TEXTURE_3D
  tex_coord_3d = tex_coord;
#endif

#ifdef USE_TEXTURE_CUBEMAP
  tex_coord_cubemap = tex_coord;
#endif

  gl_Position = projection_matrix * veye;
  #if defined(USE_IMPOSTOR_SPHERE) || defined(USE_IMPOSTOR_CYLINDER)
    // Keep the proxy alive when its center plane crosses the near plane.  The
    // fragment shader writes the analytic surface depth, so moving only the
    // proxy's clip-space depth does not change the rendered surface.
    if (gl_Position.w > 0.0 && gl_Position.z < -gl_Position.w)
      gl_Position.z = -gl_Position.w;
  #endif
}
